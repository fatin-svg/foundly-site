<script>

const examples = [

  {

    id:"demo1", type:"found", category:"Electronics",

    title:"Wireless headphones",

    description:"Wireless headphones in a dark case.",

    location:"Main library", date:"2026-10-20",

    contact:"Ask at the library desk", icon:"🎧"

  },

  {

    id:"demo2", type:"lost", category:"Bags and accessories",

    title:"Blue school backpack",

    description:"Blue backpack with a small keychain attached.",

    location:"Sports hall", date:"2026-10-19",

    contact:"Ask at the reception desk", icon:"🎒"

  },

  {

    id:"demo3", type:"found", category:"Keys and cards",

    title:"Keyring with three keys",

    description:"A set of keys on a silver keyring.",

    location:"Front reception", date:"2026-10-18",

    contact:"Ask at front reception", icon:"🔑"

  }

];


const categoryIcons = {

  "Electronics":"🎧",

  "Bags and accessories":"🎒",

  "Keys and cards":"🔑",

  "Clothing":"🧥",

  "Books and stationery":"📚",

  "Other":"📦"

};


function getSaved() {

  try {

    return JSON.parse(localStorage.getItem("foundlyReports") || "[]");

  } catch (error) {

    return [];

  }

}


function escapeHTML(value) {

  return String(value ?? "").replace(/[&<>"']/g, character => ({

    "&":"&amp;",

    "<":"&lt;",

    ">":"&gt;",

    '"':"&quot;",

    "'":"&#39;"

  })[character]);

}


function renderItems() {

  const query = document.getElementById("search").value.toLowerCase();

  const filter = document.getElementById("filter").value;

  const items = [...getSaved(), ...examples].filter(item => {

    const matchesType = filter === "all" || item.type === filter;

    const searchable = [

      item.title, item.description, item.location, item.category

    ].join(" ").toLowerCase();

    return matchesType && searchable.includes(query);

  });


  const root = document.getElementById("cards");


  if (!items.length) {

    root.innerHTML = `

      <div class="empty">

        <h3>No matching items yet</h3>

        <p>Try another search or report an item.</p>

        <button class="primary" onclick="openReport()">Report item</button>

      </div>`;

    return;

  }


  root.innerHTML = items.map(item => `

    <article class="card">

      <div class="card-top">

        <div class="emoji">${item.icon || "📦"}</div>

        <span class="tag ${item.type}">

          ${item.type === "lost" ? "LOST ITEM" : "FOUND ITEM"}

        </span>

      </div>

      <h3>${escapeHTML(item.title)}</h3>

      <p>${escapeHTML(item.description)}</p>

      <div class="meta">📍 ${escapeHTML(item.location)}</div>

      <div class="meta">

        📅 ${escapeHTML(item.date)} · ${escapeHTML(item.category)}

      </div>

      <button class="secondary" onclick="showDetails('${item.id}')">

        View details

      </button>

    </article>

  `).join("");

}


function showDetails(id) {

  const item = [...getSaved(), ...examples].find(entry => entry.id === id);

  if (!item) return;


  alert(

    item.title + "\n\n" +

    "Type: " + item.type + "\n" +

    "Location: " + item.location + "\n" +

    "Date: " + item.date + "\n\n" +

    "Contact instructions: " +

    (item.contact || "No contact instructions provided.") +

    "\n\nVerify ownership and arrange a safe handover."

  );

}


function openReport(type) {

  document.getElementById("reportModal").classList.add("open");

  document.getElementById("message").textContent = "";

  if (type) document.getElementById("reportType").value = type;

}


function closeReport() {

  document.getElementById("reportModal").classList.remove("open");

}


document.getElementById("reportForm").addEventListener("submit", event => {

  event.preventDefault();


  const form = new FormData(event.target);

  const category = form.get("category");

  const report = {

    id: "user" + Date.now(),

    type: form.get("type"),

    category,

    title: form.get("title").trim(),

    description: form.get("description").trim(),

    location: form.get("location").trim(),

    date: form.get("date"),

    contact: form.get("contact").trim(),

    icon: categoryIcons[category] || "📦"

  };


  try {

    const saved = getSaved();

    saved.unshift(report);

    localStorage.setItem("foundlyReports", JSON.stringify(saved));


    event.target.reset();

    closeReport();

    renderItems();


    document.getElementById("browse").scrollIntoView();

    alert("Your report has been saved in this browser!");

  } catch (error) {

    alert("The report could not be saved. Check your browser settings.");

  }

});


document.getElementById("reportModal").addEventListener("click", event => {

  if (event.target.id === "reportModal") closeReport();

});


document.addEventListener("keydown", event => {

  if (event.key === "Escape") closeReport();

});


renderItems();

</script>
